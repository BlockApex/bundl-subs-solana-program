use anchor_lang::prelude::*;
use solana_program::keccak;

declare_id!("35YWrusRg5PBtjKvrKjckewYrfTwGEEx1rqTzwPkTRzq");

const OWNER: &str = "BTFsHVsT8V9gXrgDNKdCw574dR9X8hom9KWsiKBvjbSi";

#[program]
pub mod referrals {
    use super::*;

    #[access_control(check_owner(&ctx.accounts.authority))]
    pub fn initialize(ctx: Context<Initialize>, merkle_root: [u8; 32]) -> Result<()> {
        let merkle_tree = &mut ctx.accounts.merkle_tree;
        merkle_tree.merkle_root = merkle_root;
        merkle_tree.claimed_users = Vec::new();  // Initialize the claimed_users vector
        msg!("Initialized merkle tree with root: {:?}", merkle_root);
        Ok(())
    }

    #[access_control(check_owner(&ctx.accounts.authority))]
    pub fn update_merkle_root(
        ctx: Context<UpdateMerkleRoot>,
        new_merkle_root: [u8; 32],
    ) -> Result<()> {
        let merkle_tree = &mut ctx.accounts.merkle_tree;
        merkle_tree.merkle_root = new_merkle_root;
        merkle_tree.claimed_users.clear();  // Clear the claimed_users vector
        msg!("Updated merkle root to: {:?}", new_merkle_root);
        Ok(())
    }

    pub fn claim(
        ctx: Context<Claim>,
        user_address: Pubkey,
        amount: u64,
        proof: Vec<[u8; 32]>,
    ) -> Result<()> {
        let merkle_tree = &ctx.accounts.merkle_tree;

        // Check if the user has already claimed
        require!(
            !merkle_tree.claimed_users.contains(&user_address),
            ErrorCode::AlreadyClaimed
        );

        // Check that the signer matches the user_address
        require_keys_eq!(
            ctx.accounts.user.key(),
            user_address,
            ErrorCode::UserAddressMismatch
        );
        msg!("User address check passed for: {:?}", user_address);

        // Convert amount to string
        let amount_str = amount.to_string();

        // Combine user_address_bytes and amount_str into a single string
        let combined = format!("{:?}{}", user_address, amount_str);

        // Convert combined string to bytes
        let combined_bytes = combined.as_bytes();

        // Hash the combined byte slice
        let hash = keccak::hash(combined_bytes);
        msg!("Initial hash: {:?}", hash);

        let mut final_computed_hash = hash; // Initialize final_computed_hash

        for p in proof.iter() {
            let proof_node = keccak::Hash::new_from_array(*p);
            msg!("Processing proof node: {:?}", proof_node);

            if final_computed_hash.to_bytes() <= proof_node.to_bytes() {
                final_computed_hash =
                    keccak::hashv(&[final_computed_hash.as_ref(), proof_node.as_ref()]);
            } else {
                final_computed_hash =
                    keccak::hashv(&[proof_node.as_ref(), final_computed_hash.as_ref()]);
            }
            msg!("Updated final_computed_hash: {:?}", final_computed_hash);
        }

        require!(
            final_computed_hash.to_bytes() == merkle_tree.merkle_root,
            ErrorCode::InvalidProof
        );
        msg!(
            "Merkle proof is valid. Claim successful for user: {:?}, amount: {}",
            user_address,
            amount
        );

        // Add the user to the claimed_users vector
        let merkle_tree = &mut ctx.accounts.merkle_tree;
        merkle_tree.claimed_users.push(user_address);

        Ok(())
    }
}

fn check_owner(authority: &Signer) -> Result<()> {
    let owner_pubkey = OWNER.parse::<Pubkey>().unwrap();
    require_keys_eq!(authority.key(), owner_pubkey, CustomError::NotOwner);
    msg!("Owner check passed for authority: {:?}", authority.key());

    Ok(())
}

#[derive(Accounts)]
pub struct Initialize<'info> {
    #[account(init, payer = authority, space = 8 + 8 + 32 + 4 + 32 * 100, seeds = [b"merkle_tree_ne"], bump)]
    pub merkle_tree: Account<'info, MerkleTree>,
    #[account(mut)]
    pub authority: Signer<'info>,
    pub system_program: Program<'info, System>,
}

#[derive(Accounts)]
pub struct UpdateMerkleRoot<'info> {
    #[account(mut, seeds = [b"merkle_tree_ne"], bump)]
    pub merkle_tree: Account<'info, MerkleTree>,
    #[account(mut)]
    pub authority: Signer<'info>,
}

#[derive(Accounts)]
pub struct Claim<'info> {
    #[account(mut, seeds = [b"merkle_tree_ne"], bump)]
    pub merkle_tree: Account<'info, MerkleTree>,
    #[account(mut)]
    pub user: Signer<'info>,
}

#[account]
#[derive(InitSpace)]
pub struct MerkleTree {
    pub merkle_root: [u8; 32],
    #[max_len(100)]
    pub claimed_users: Vec<Pubkey>,  // Add a vector to track claimed users
}

#[error_code]
pub enum ErrorCode {
    #[msg("Invalid Merkle proof")]
    InvalidProof,
    #[msg("Invalid User Address")]
    UserAddressMismatch,
    #[msg("User has already claimed")]
    AlreadyClaimed,
}

#[error_code]
pub enum CustomError {
    #[msg("Only owner can call this function!")]
    NotOwner,
}
